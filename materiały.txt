12. wyświetl hobby które wystęoują u co najmniej 10 uczniów;

SELECT hobby
	COUNT(*) AS liczba_uczniow
FROM uczniowie 
GROUP BY hobby
HAVING COUNT(*) >=10
ORDER BY liczba_uczniow DESC;
